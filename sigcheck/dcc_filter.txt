#   =全名  = 精确匹配转译
#   正则   = 匹配即转译

# 排除所有构造器与类初始化器（风险高、易编译失败）
!<clinit|init>

# 只转译 MainActivity.onCreate —— 注入目标的唯一入口
# ⚠️ dcc full_name 无 "->"!真实格式: 类名;方法名(原型)，由 ''.join(method_triple) 拼接
# 例: Lcom/xixin/box/MainActivity;onCreate(Landroid/os/Bundle;)V
L.*MainActivity;onCreate\(Landroid/os/Bundle;\)
